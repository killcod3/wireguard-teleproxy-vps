#!/bin/bash
set -euo pipefail
export DEBIAN_FRONTEND=noninteractive

TELEPROXY_PUBLIC_PORT="443"
TELEPROXY_PROXY_PORT="8888"

apt-get update
apt-get install -y curl ca-certificates openssl

detect_public_ip() {
  local ip=""
  local token=""

  ip="$(curl -4 -fsS --connect-timeout 2 http://169.254.169.254/metadata/v1/interfaces/public/0/ipv4/address 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  ip="$(curl -4 -fsS --connect-timeout 5 https://api.ipify.org 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  ip="$(curl -4 -fsS --connect-timeout 5 https://ifconfig.me 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  token="$(curl -fsS --connect-timeout 2 -X PUT http://169.254.169.254/latest/api/token -H 'X-aws-ec2-metadata-token-ttl-seconds: 60' 2>/dev/null || true)"
  if [ -n "$token" ]; then
    ip="$(curl -4 -fsS --connect-timeout 2 -H "X-aws-ec2-metadata-token: $token" http://169.254.169.254/latest/meta-data/public-ipv4 2>/dev/null || true)"
    if [ -n "$ip" ]; then echo "$ip"; return 0; fi
  fi

  ip="$(curl -4 -fsS --connect-timeout 2 http://169.254.169.254/latest/meta-data/public-ipv4 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  return 1
}

SERVER_IP="$(detect_public_ip || true)"

if [ -z "${SERVER_IP}" ]; then
  echo "ERROR: Could not detect public IPv4 address"
  exit 1
fi

useradd --system --no-create-home --shell /usr/sbin/nologin teleproxy || true

curl -fsSL https://github.com/teleproxy/teleproxy/releases/latest/download/teleproxy-linux-amd64 -o /usr/local/bin/teleproxy
chmod 711 /usr/local/bin/teleproxy

umask 077

mkdir -p /etc/teleproxy
SECRET="$(openssl rand -hex 16)"
openssl rand -hex 32 > /etc/teleproxy/aes-pwd.txt

cat > /etc/teleproxy/teleproxy.env <<EOF
SECRET=${SECRET}
TELEPROXY_PUBLIC_PORT=${TELEPROXY_PUBLIC_PORT}
TELEPROXY_PROXY_PORT=${TELEPROXY_PROXY_PORT}
EOF

chown -R teleproxy:teleproxy /etc/teleproxy
chmod 700 /etc/teleproxy
chmod 600 /etc/teleproxy/teleproxy.env /etc/teleproxy/aes-pwd.txt

cat > /etc/systemd/system/teleproxy.service <<'EOF'
[Unit]
Description=Teleproxy MTProto Proxy
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=teleproxy
EnvironmentFile=/etc/teleproxy/teleproxy.env
ExecStart=/usr/local/bin/teleproxy -S ${SECRET} -H ${TELEPROXY_PUBLIC_PORT} --direct -p ${TELEPROXY_PROXY_PORT} --aes-pwd /etc/teleproxy/aes-pwd.txt
Restart=on-failure
RestartSec=5
LimitNOFILE=65536
AmbientCapabilities=CAP_NET_BIND_SERVICE
CapabilityBoundingSet=CAP_NET_BIND_SERVICE
NoNewPrivileges=true

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now teleproxy

cat > /root/TELEPROXY-VPS-INFO.txt <<EOF
Server public IP: ${SERVER_IP}

Teleproxy:
Public listen port: ${TELEPROXY_PUBLIC_PORT}/tcp
Proxy option port: ${TELEPROXY_PROXY_PORT}
Secret: ${SECRET}

Firewall allow:
22/tcp
443/tcp

Teleproxy link:
https://t.me/proxy?server=${SERVER_IP}&port=${TELEPROXY_PUBLIC_PORT}&secret=${SECRET}

Checks:
systemctl is-active teleproxy
systemctl status teleproxy --no-pager -l
ss -tlpn | grep ':443'
journalctl -u teleproxy -n 50 --no-pager -l
cat /var/log/cloud-init-output.log
EOF

chmod 600 /root/TELEPROXY-VPS-INFO.txt
