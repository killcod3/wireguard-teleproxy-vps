#!/bin/bash
set -euo pipefail
export DEBIAN_FRONTEND=noninteractive

WG_PORT="51820"
WG_NET="10.8.0.0/24"
SERVER_VPN_IP="10.8.0.1"
CLIENT_COUNT="5"

apt-get update
apt-get install -y wireguard qrencode iptables curl ca-certificates

detect_public_ip() {
  local ip=""
  local token=""

  ip="$(curl -4 -fsS --connect-timeout 2 http://169.254.169.254/metadata/v1/interfaces/public/0/ipv4/address 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  ip="$(curl -4 -fsS --connect-timeout 5 https://api.ipify.org 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  ip="$(curl -4 -fsS --connect-timeout 5 https://ifconfig.me 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  token="$(curl -fsS --connect-timeout 2 -X PUT http://169.254.169.254/latest/api/token -H 'X-aws-ec2-metadata-token-ttl-seconds: 60' 2>/dev/null || true)"
  if [ -n "$token" ]; then
    ip="$(curl -4 -fsS --connect-timeout 2 -H "X-aws-ec2-metadata-token: $token" http://169.254.169.254/latest/meta-data/public-ipv4 2>/dev/null || true)"
    if [ -n "$ip" ]; then echo "$ip"; return 0; fi
  fi

  ip="$(curl -4 -fsS --connect-timeout 2 http://169.254.169.254/latest/meta-data/public-ipv4 2>/dev/null || true)"
  if [ -n "$ip" ]; then echo "$ip"; return 0; fi

  return 1
}

SERVER_IP="$(detect_public_ip || true)"

if [ -z "${SERVER_IP}" ]; then
  echo "ERROR: Could not detect public IPv4 address"
  exit 1
fi

IFACE="$(ip route get 1.1.1.1 | awk '{for(i=1;i<=NF;i++) if($i=="dev") {print $(i+1); exit}}' || true)"

if [ -z "${IFACE}" ]; then
  echo "ERROR: Could not detect default network interface"
  exit 1
fi

mkdir -p /etc/wireguard/clients
cd /etc/wireguard
umask 077

wg genkey | tee server_private.key | wg pubkey > server_public.key

cat > /etc/wireguard/wg0.conf <<EOF
[Interface]
Address = ${SERVER_VPN_IP}/24
ListenPort = ${WG_PORT}
PrivateKey = $(cat server_private.key)
PostUp = sysctl -w net.ipv4.ip_forward=1; iptables -A FORWARD -i wg0 -j ACCEPT; iptables -A FORWARD -o wg0 -j ACCEPT; iptables -t nat -A POSTROUTING -s ${WG_NET} -o ${IFACE} -j MASQUERADE
PostDown = iptables -D FORWARD -i wg0 -j ACCEPT; iptables -D FORWARD -o wg0 -j ACCEPT; iptables -t nat -D POSTROUTING -s ${WG_NET} -o ${IFACE} -j MASQUERADE
EOF

for i in $(seq 1 "$CLIENT_COUNT"); do
  CLIENT_IP="10.8.0.$((i+1))"

  wg genkey | tee "clients/client${i}_private.key" | wg pubkey > "clients/client${i}_public.key"

  cat >> /etc/wireguard/wg0.conf <<EOF

[Peer]
# client${i}
PublicKey = $(cat clients/client${i}_public.key)
AllowedIPs = ${CLIENT_IP}/32
EOF

  cat > "/etc/wireguard/clients/client${i}.conf" <<EOF
[Interface]
PrivateKey = $(cat clients/client${i}_private.key)
Address = ${CLIENT_IP}/32
DNS = 1.1.1.1

[Peer]
PublicKey = $(cat server_public.key)
Endpoint = ${SERVER_IP}:${WG_PORT}
AllowedIPs = 0.0.0.0/0
PersistentKeepalive = 25
EOF

  qrencode -t ansiutf8 < "/etc/wireguard/clients/client${i}.conf" > "/etc/wireguard/clients/client${i}-qr.txt"
done

chmod 600 /etc/wireguard/wg0.conf /etc/wireguard/*key /etc/wireguard/clients/*

cat > /etc/sysctl.d/99-wireguard.conf <<EOF
net.ipv4.ip_forward=1
EOF

sysctl --system
systemctl enable --now wg-quick@wg0

cat > /root/WIREGUARD-VPS-INFO.txt <<EOF
Server public IP: ${SERVER_IP}
Default interface: ${IFACE}

WireGuard:
Public port: ${WG_PORT}/udp
Client configs: /etc/wireguard/clients/client1.conf ... client5.conf
QR files: /etc/wireguard/clients/client1-qr.txt ... client5-qr.txt

Firewall allow:
22/tcp
51820/udp

Checks:
systemctl is-active wg-quick@wg0
wg show
ss -ulpn | grep ':51820'
cat /var/log/cloud-init-output.log
EOF
